import cv2

cap = cv2.VideoCapture(0)  # Try the default camera

if not cap.isOpened():
    print("Camera is not accessible!")
else:
    print("Camera is working!")
    while True:
        ret, frame = cap.read()
        if not ret:
            print("Failed to grab frame!")
            break
        cv2.imshow("Camera", frame)
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break

cap.release()
cv2.destroyAllWindows()
